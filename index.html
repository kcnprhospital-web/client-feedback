🔗 Google Apps Script Setup Guide

Step 1: Create a new Google Sheet named "KCPH_Feedback"
Step 2: Extensions → Apps Script → paste code below
Step 3: Deploy → New deployment → Web App → Execute as "Me" → Anyone
Step 4: Copy deployment URL → paste in Configuration above

const SHEET_NAME = 'Feedback'; const BACKUP_SHEET = 'Backup_Raw'; function doPost(e) { try { const data = JSON.parse(e.postData.contents); const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) sheet = ss.insertSheet(SHEET_NAME); if (sheet.getLastRow() === 0) { sheet.appendRow(['Timestamp','Payment Method','Type','Department', 'Q1','Q2','Q3','Q4','Q5','Q6','Avg Score','Comment']); sheet.setFrozenRows(1); } if (data.action === 'clearAll') { if (sheet.getLastRow() > 1) sheet.deleteRows(2, sheet.getLastRow()-1); return ContentService.createTextOutput(JSON.stringify({success:true})) .setMimeType(ContentService.MimeType.JSON); } const r = data.ratings || {}; const keys = Object.keys(r); sheet.appendRow([ new Date(data.timestamp).toISOString(), data.paymentMethod || '', data.type || '', data.department || '', r[keys[0]]||'', r[keys[1]]||'', r[keys[2]]||'', r[keys[3]]||'', r[keys[4]]||'', r[keys[5]]||'', parseFloat(data.avgScore)||0, data.comment || '' ]); let bk = ss.getSheetByName(BACKUP_SHEET); if (!bk) { bk = ss.insertSheet(BACKUP_SHEET); bk.appendRow(['Timestamp','Raw JSON']); bk.setFrozenRows(1); } bk.appendRow([new Date().toISOString(), JSON.stringify(data)]); return ContentService.createTextOutput(JSON.stringify({success:true})) .setMimeType(ContentService.MimeType.JSON); } catch(err) { try { var ss2=SpreadsheetApp.getActiveSpreadsheet(); var el=ss2.getSheetByName('ErrorLog')||ss2.insertSheet('ErrorLog'); el.appendRow([new Date().toISOString(),err.message]); } catch(x){} return ContentService.createTextOutput(JSON.stringify({error:err.message})) .setMimeType(ContentService.MimeType.JSON); } } function doGet(e) { try { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) return out([]); const rows = sheet.getDataRange().getValues(); if (rows.length <= 1) return out([]); function toISO(val) { if (!val) return ''; if (val instanceof Date) return val.toISOString(); var s = String(val).trim(); var m = s.match(/^(\d{1,2})\/(\d{1,2})\/(\d{4})(?:\s(\d{2}:\d{2}(?::\d{2})?))?/); if (m) try { return new Date(m[3]+'-'+m[2].padStart(2,'0')+'-'+m[1].padStart(2,'0')+'T'+(m[4]||'00:00:00')+'+07:00').toISOString(); } catch(x){} try { var d=new Date(s); if(!isNaN(d.getTime())) return d.toISOString(); } catch(x){} return ''; } const DEPTS = ['OPD','Emergency','ICU','General Medicine','Trauma-Surgery A','Trauma-Surgery B', 'Operation Theater','Gynecology','Maternity','Pediatry','NCU','Eye Unit','Dialysis','Medicine A', 'Laboratory','Blood Bank','Radiology','Ultrasound','PMRS','Cashier','Pharmacy']; const PAYS = ['self','hef','nssf','exempt']; const TYPES = ['care','support']; function isDept(v) { var vl=String(v||'').trim().toLowerCase(); return DEPTS.some(function(d){return d.toLowerCase()===vl||vl.startsWith(d.toLowerCase());}); } function isType(v) { return TYPES.indexOf(String(v||'').trim().toLowerCase())>=0; } function isPay(v) { return PAYS.indexOf(String(v||'').trim().toLowerCase())>=0; } function normDept(v) { var s=String(v||'').trim(); if(s.indexOf('/')>0){var e2=s.split('/')[0].trim();if(/[a-zA-Z]/.test(e2))s=e2;} var sl=s.toLowerCase(); for(var i=0;i<DEPTS.length;i++){if(DEPTS[i].toLowerCase()===sl)return DEPTS[i];} return s; } var data = rows.slice(1).map(function(row) { // Read raw column values by position (0-based) // Possible formats in the sheet: // NEW: [Timestamp, PayMethod, Type, Dept, Q1-Q6, AvgScore, Comment] = 12 cols // OLD: [Timestamp, Type, Dept, Q1-Q6, AvgScore] = 11 cols (no payment, no comment) var ts = toISO(row[0]); if (!ts) return null; var c1 = String(row[1]||'').trim(); // col B var c2 = String(row[2]||'').trim(); // col C var c3 = String(row[3]||'').trim(); // col D var pay='', type='', dept='', avg=0, comment=''; if (isType(c2) && isDept(c3)) { // NEW format: B=payment, C=type, D=dept pay = isPay(c1.toLowerCase()) ? c1.toLowerCase() : ''; type = c2.toLowerCase(); dept = normDept(c3); avg = parseFloat(row[10])||0; // col K = Avg Score comment = String(row[11]===undefined||row[11]===null?'':row[11]).trim(); // col L = Comment } else if (isType(c1) && isDept(c2)) { // OLD format: B=type, C=dept (no payment column) pay = ''; type = c1.toLowerCase(); dept = normDept(c2); avg = parseFloat(row[9])||0; // col J = Avg Score (old format) comment = String(row[10]===undefined||row[10]===null?'':row[10]).trim(); // col K = Comment (old) } else { // Fallback: scan for dept pay=''; type='care'; dept=''; avg=0; comment=''; for(var i=1;i<row.length;i++){ if(!dept&&isDept(row[i])){dept=normDept(row[i]);} } // Avg Score is always in col K (index 10) for new rows avg = parseFloat(row[10])||parseFloat(row[9])||0; comment = String(row[11]===undefined||row[11]===null?'':row[11]).trim(); } if (!dept) return null; return { 'Timestamp': ts, 'Payment Method': pay, 'Type': type||'care', 'Department': dept, 'Q1':String(row[4]||''), 'Q2':String(row[5]||''), 'Q3':String(row[6]||''), 'Q4':String(row[7]||''), 'Q5':String(row[8]||''), 'Q6':isType(c2)&&isDept(c3)?String(row[9]||''):'', 'Avg Score': avg, 'Comment': comment }; }).filter(function(r){return r!==null;}); return out(data); } catch(err) { return out({error:err.message}); } } function out(data) { return ContentService.createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); } function weeklyBackup() { const ss = SpreadsheetApp.getActiveSpreadsheet(); const src = ss.getSheetByName(SHEET_NAME); if (!src) return; const date = Utilities.formatDate(new Date(),'Asia/Phnom_Penh','yyyy-MM-dd'); const copy = src.copyTo(ss); copy.setName('Backup_'+date); const sheets = ss.getSheets().filter(function(s){return s.getName().startsWith('Backup_20');}); if(sheets.length>8){sheets.sort(function(a,b){return a.getName().localeCompare(b.getName());});ss.deleteSheet(sheets[0]);} }
